def twoSum(nums, target):
    d={}
    for i in range(len(nums)):
        if target-nums[i] in d:
            return [d[target-nums[i]],i]
        d[nums[i]]=i

 def longestConsecutive(arr):
    s=set(arr); longest=0
    for x in s:
        if x-1 not in s:
            y=x
            while y in s: y+=1
            longest=max(longest,y-x)
    return longest
def setZero(mat):
    r=set(); c=set()
    for i in range(len(mat)):
        for j in range(len(mat[0])):
            if mat[i][j]==0:
                r.add(i); c.add(j)
    for i in r: mat[i]=[0]*len(mat[0])
    for j in c:
        for i in range(len(mat)):
            mat[i][j]=0
    return mat

    temp=head
    if temp and temp.data==key:
        return temp.next
